lib/main.dart — ShellRoute & Deep Link Configuration
final GoRouter appRouter = GoRouter(
  initialLocation: '/home',
  routes: [
    // 1. Nested Navigation Flow via ShellRoute
    ShellRoute(
      builder: (context, state, child) => ScaffoldWithNavBar(child: child),
      routes: [
        GoRoute(path: '/home', name: 'home', builder: (_, __) => const HomeScreen()),
        GoRoute(path: '/explore', name: 'explore', builder: (_, __) => const ExploreScreen()),
        GoRoute(path: '/settings', name: 'settings', builder: (_, __) => const SettingsScreen()),
      ],
    ),
    // 2. Parameterized Deep Link Route
    GoRoute(
      path: '/item/:id',
      name: 'item-details',
      builder: (context, state) {
        final id = state.pathParameters['id'] ?? 'unknown';
        final name = state.uri.queryParameters['name'] ?? 'Custom Item';
        return ItemDetailScreen(itemId: id, itemName: name);
      },
    ),
    // 3. Additional Named Route
    GoRoute(path: '/about', name: 'about', builder: (_, __) => const AboutScreen()),
  ],
);
